cmake_minimum_required(VERSION 3.21)
project(AriaDownload VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.2 REQUIRED COMPONENTS Widgets Network)
add_library(aria_backend src/AriaBackend.cpp src/AriaBackend.h)
target_link_libraries(aria_backend PUBLIC Qt6::Network)
target_include_directories(aria_backend PUBLIC src)
qt_add_executable(AriaDownload WIN32 src/main.cpp src/MainWindow.cpp src/MainWindow.h src/DownloadModel.cpp src/DownloadModel.h)
target_link_libraries(AriaDownload PRIVATE aria_backend Qt6::Widgets)
if(MSVC)
  target_compile_options(aria_backend PRIVATE /W4 /utf-8)
  target_compile_options(AriaDownload PRIVATE /W4 /utf-8)
endif()
include(CTest)
if(BUILD_TESTING)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  qt_add_executable(backend_tests tests/backend_tests.cpp)
  target_link_libraries(backend_tests PRIVATE aria_backend Qt6::Test)
  set_target_properties(backend_tests PROPERTIES WIN32_EXECUTABLE FALSE)
  add_test(NAME backend_integration COMMAND backend_tests -o backend-results.txt,txt)
  set_tests_properties(backend_integration PROPERTIES TIMEOUT 120)
  qt_add_executable(gui_tests tests/gui_tests.cpp src/MainWindow.cpp src/MainWindow.h src/DownloadModel.cpp src/DownloadModel.h)
  target_link_libraries(gui_tests PRIVATE aria_backend Qt6::Widgets Qt6::Test)
  set_target_properties(gui_tests PROPERTIES WIN32_EXECUTABLE FALSE)
  if(MSVC)
    target_compile_options(gui_tests PRIVATE /utf-8)
  endif()
  add_test(NAME gui_smoke COMMAND gui_tests -platform offscreen -o gui-results.txt,txt)
  set_tests_properties(gui_smoke PROPERTIES TIMEOUT 30)
endif()
install(TARGETS AriaDownload RUNTIME DESTINATION .)
